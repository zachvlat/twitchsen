Export your Twitch following list

On twitch.tv, open your Following page, then run this in the browser console (DevTools → Console). It downloads a JSON file you can import here.

const values = [...document.querySelectorAll(
  'div.Layout-sc-1xcs6mc-0 > div > div > div > div:nth-child(3) > div > div > a > strong:nth-child(2)'
)].map(el => el.textContent.trim());

const json = JSON.stringify(values, null, 2);
const blob = new Blob([json], { type: "application/json" });
const url = URL.createObjectURL(blob);

const a = document.createElement("a");
a.href = url;
a.download = "following.json";
a.click();

URL.revokeObjectURL(url);
